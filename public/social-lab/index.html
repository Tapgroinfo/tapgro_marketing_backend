Phase 3 pattern (inside Nest)

              
// Prefer existing wrappers:
await this.zernio.createProfile({ name, description, externalId });
await this.zernio.getConnectUrl('instagram', { profileId, redirect_url });

// Or raw SDK for new surfaces (comments / DMs):
const sdk = await this.zernio.getSdk();
const { data } = await sdk.comments.listInboxComments({
  query: { profileId, limit: 20 },
});
